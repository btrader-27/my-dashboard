Handles Tasks and Projects from your Google Sheet
Your Boris Schlossberg and Boris Work calendars are pulled directly by the Apps Script using your Google account. Just make sure you deploy the script as Execute as: Me.
📋 Apps Script — paste this entire block (replaces all existing code)
// ── CALENDAR NAMES TO PULL ────────────────────
var CAL_NAMES = ["Boris Schlossberg", "Boris Work"];
var CAL_DAYS_BACK   = 7;   // days in the past to show
var CAL_DAYS_FORWARD = 60; // days ahead to show

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // ── TOGGLE TASK DONE ──────────────────────────
  if (e.parameter.action === "toggle") {
    var sheet = ss.getSheetByName("Tasks");
    var taskName = e.parameter.task;
    var newDone  = e.parameter.done === "true";
    var rows = sheet.getDataRange().getValues();
    var hi = -1;
    for (var i = 0; i < rows.length; i++) {
      if (rows[i].map(c=>String(c).toLowerCase())
          .indexOf("task") !== -1) { hi=i; break; }
    }
    var hdrs = rows[hi].map(c=>String(c).toLowerCase());
    var tc = hdrs.indexOf("task");
    var dc = hdrs.indexOf("done?");
    if (dc===-1) dc = hdrs.indexOf("done");
    for (var r = hi+1; r < rows.length; r++) {
      if (String(rows[r][tc]).trim() === taskName) {
        sheet.getRange(r+1, dc+1)
          .setValue(newDone ? "Yes" : "No");
        break;
      }
    }
    return ContentService.createTextOutput("ok")
      .setMimeType(ContentService.MimeType.TEXT);
  }

  // ── READ ALL DATA ─────────────────────────────
  var out = { tasks: [], projects: [], calEvents: [] };

  // Tasks tab
  var ts = ss.getSheetByName("Tasks");
  if (ts) {
    var trows = ts.getDataRange().getValues();
    var thi = -1;
    for (var i=0;i<trows.length;i++){
      if(trows[i].map(c=>String(c).toLowerCase()).indexOf("task")!==-1){thi=i;break;}
    }
    if (thi !== -1) {
      var th = trows[thi].map(c=>String(c).toLowerCase());
      for (var r=thi+1;r<trows.length;r++){
        var row=trows[r]; var obj={};
        th.forEach((h,i)=>{obj[h]=String(row[i]||"").trim();});
        if(obj.task) out.tasks.push(obj);
      }
    }
  }

  // Projects tab
  var ps = ss.getSheetByName("Projects");
  if (ps) {
    var prows = ps.getDataRange().getValues();
    var headers = prows[0].map(c=>String(c).toLowerCase());
    var projects = {};
    for (var r=1;r<prows.length;r++){
      var row=prows[r]; var obj={};
      headers.forEach((h,i)=>{obj[h]=String(row[i]||"").trim();});
      if (!obj.project) continue;
      if (obj.type==="project" || !obj.type) {
        projects[obj.project]={
          name:obj.project, status:obj.status||"",
          deadline:obj.deadline||"", notes:obj.notes||"",
          subtasks:[]
        };
      } else if (obj.type==="subtask") {
        if (!projects[obj.project])
          projects[obj.project]={name:obj.project,status:"",deadline:"",notes:"",subtasks:[]};
        projects[obj.project].subtasks.push({
          name:obj.task||obj.name||"", status:obj.status||""
        });
      }
    }
    out.projects = Object.values(projects);
  }

  // Google Calendar — pulled directly via CalendarApp
  var now   = new Date();
  var start = new Date(now); start.setDate(start.getDate() - CAL_DAYS_BACK);
  var end   = new Date(now); end.setDate(end.getDate() + CAL_DAYS_FORWARD);

  CAL_NAMES.forEach(function(calName) {
    var cal = CalendarApp.getCalendarsByName(calName)[0];
    if (!cal) return;
    var events = cal.getEvents(start, end);
    events.forEach(function(ev) {
      var sd = ev.getStartTime();
      var allDay = ev.isAllDayEvent();
      var dateStr = Utilities.formatDate(sd, Session.getScriptTimeZone(), "yyyy-MM-dd");
      var timeStr = allDay ? "" :
        Utilities.formatDate(sd, Session.getScriptTimeZone(), "h:mm a");
      out.calEvents.push({
        title:    ev.getTitle(),
        date:     dateStr,
        time:     timeStr,
        notes:    ev.getDescription() || "",
        location: ev.getLocation() || "",
        src:      calName === "Boris Work" ? "gcal-work" : "gcal-personal",
        calName:  calName
      });
    });
  });

  // Calendar Sheet tab (manual entries — merged in)
  var cs = ss.getSheetByName("Calendar");
  if (cs) {
    var crow = cs.getDataRange().getValues();
    var ch = crow[0].map(c=>String(c).toLowerCase());
    for (var r=1;r<crow.length;r++){
      var obj={};
      ch.forEach((h,i)=>{obj[h]=String(crow[r][i]||"").trim();});
      if(obj.title||obj.event) out.calEvents.push({
        title:obj.title||obj.event, date:obj.date||"",
        time:obj.time||"", notes:obj.notes||"", src:"sheet"
      });
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify(out))
    .setMimeType(ContentService.MimeType.JSON);
}
