function doGet(e) {
  var sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName("Tasks");
  var rows = sheet.getDataRange().getValues();
  var csv = rows.map(r => r.join(",")).join("\n");
  return ContentService.createTextOutput(csv)
    .setMimeType(ContentService.MimeType.TEXT);
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName("Tasks");
  var rows = sheet.getDataRange().getValues();
  var headerIdx = -1;
  for (var i = 0; i < rows.length; i++) {
    var low = rows[i].map(c => String(c).toLowerCase());
    if (low.indexOf("task") !== -1) { headerIdx = i; break; }
  }
  var hdrs = rows[headerIdx].map(c => String(c).toLowerCase());
  var taskCol = hdrs.indexOf("task");
  var doneCol = hdrs.indexOf("done?");
  if (doneCol === -1) doneCol = hdrs.indexOf("done");
  for (var r = headerIdx + 1; r < rows.length; r++) {
    if (String(rows[r][taskCol]).trim() === data.task) {
      sheet.getRange(r + 1, doneCol + 1)
        .setValue(data.done ? "Yes" : "No");
      break;
    }
  }
  return ContentService.createTextOutput("ok")
    .setMimeType(ContentService.MimeType.TEXT);
}
